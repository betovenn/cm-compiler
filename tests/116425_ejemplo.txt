int main() {
    int filasA;
    int columnasA; 
    int filasB; 
    int columnasB;

    // Validar si se pueden multiplicar
    if (columnasA != filasB) {
        return 1;
    }

    // Declarar las matrices
    int A; 
    int B; 
    int C;

    // Rellenar la Matriz A
    for (int i = 0; i < filasA; i=i+1) {
        for (int j = 0; j < columnasA; j++) {
            A=1;
        }
    }

    // Rellenar la Matriz B
    for (int i = 0; i < filasB; i++) {
        for (int j = 0; j < columnasB; j++) {
            B=1;
        }
    }

    // Inicializar la matriz resultante C en 0
    for (int i = 0; i < filasA; i++) {
        for (int j = 0; j < columnasB; j++) {
            C = 0;
        }
    }

    // Multiplicar las matrices
    for (int i = 0; i < filasA; i++) {
        for (int j = 0; j < columnasB; j++) {
            for (int k = 0; k < columnasA; k++) {
                C = A * B;
            }
        }
    }


    return 0;
}
